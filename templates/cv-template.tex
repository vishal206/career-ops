%  SECTION HEADING -- lowercase \section retains the validator
%  contract; restyled to the bold title + inline gray rule look.
% ============================================================
\renewcommand{\section}[1]{%
  \vspace{\sectionSpaceBefore}\noindent
  \makebox[\linewidth][l]{%
    \textbf{\large #1}\hspace{0.6em}%
    \textcolor{gray!50}{\leaders\hrule height 0.30pt\hfill\kern0pt}%
  }%
  \par\vspace{\sectionSpaceAfter}%
}

% ============================================================
%  RESUME COMMANDS (contract used by build-cv-latex.mjs)
% ============================================================
% Outer wrapper around a group of entries.
\newcommand{\resumeSubHeadingListStart}{%
  \begin{itemize}[leftmargin=0pt, label={}, topsep=0pt, partopsep=0pt, parsep=0pt, itemsep=\entryGap]%
}
\newcommand{\resumeSubHeadingListEnd}{\end{itemize}}

% #1 top-left (company / institution), #2 top-right (dates / location),
% #3 second-left (role / degree),      #4 second-right (location / dates).
\newcommand{\resumeSubheading}[4]{%
  \item
  \begin{tabular*}{\linewidth}[t]{@{}l@{\extracolsep{\fill}}r@{}}
    \textbf{\textcolor{subTextColor}{#1}} & {\small\textcolor{secondaryColor}{#2}} \\[\titleToCompany]
    {\small\textcolor{secondaryColor}{#3}} & {\small\textcolor{secondaryColor}{#4}} \\
  \end{tabular*}%
  \vspace{\headerToBullets}%
}

% #1 pre-built "\textbf{name} \emph{$|$ context}", #2 dates (may be empty).
\newcommand{\resumeProjectHeading}[2]{%
  \item
  \begin{tabular*}{\linewidth}[t]{@{}l@{\extracolsep{\fill}}r@{}}
    \textcolor{subTextColor}{#1} & {\small\textcolor{secondaryColor}{#2}} \\
  \end{tabular*}%
  \vspace{\headerToBullets}%
}

\newcommand{\resumeItem}[1]{\item\small #1}
\newcommand{\resumeItemListStart}{%
  \begin{itemize}[topsep=\bulletTopSep, partopsep=0pt, parsep=0pt, itemsep=\bulletItemSep, leftmargin=11pt]%
}
\newcommand{\resumeItemListEnd}{\end{itemize}}
